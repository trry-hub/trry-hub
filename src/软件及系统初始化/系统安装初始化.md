export NVM_DIR="$HOME/.nvm"
[ -s "$NVM_DIR/nvm.sh" ] && \. "$NVM_DIR/nvm.sh"

### 最优方案：智能版本切换（推荐）

将以下代码添加到 `~/.zshrc`：

```bash
# =============================================================================
# 自动切换 Node.js 版本 (基于 package.json engines.node)
# =============================================================================

# 自动切换 Node.js 版本函数
auto_switch_node() {
  # 检查是否存在 package.json
  if [[ ! -f package.json ]]; then
    return 0
  fi

  # 检查 nvm 是否可用
  if ! command -v nvm >/dev/null 2>&1; then
    echo -e "\033[33m⚠️  nvm 未安装，无法自动切换 Node.js 版本\033[0m" >&2
    return 1
  fi

  # 获取当前 Node.js 版本
  local current_version=$(node -v 2>/dev/null | sed 's/^v//' || echo "")
  
  # 解析 package.json 中的 engines.node
  local node_range=$(node -e "
    try {
      const fs = require('fs');
      const pkg = JSON.parse(fs.readFileSync('package.json', 'utf8'));
      console.log(pkg.engines?.node || '');
    } catch (e) {
      console.log('');
    }
  " 2>/dev/null)

  # 如果没有指定版本范围，直接返回
  if [[ -z "$node_range" ]]; then
    return 0
  fi

  # 简化版本范围解析 - 处理常见的格式
  local target_version=""
  
  # 对于 >=X.Y.Z 格式
  if [[ "$node_range" =~ ^\>\=[[:space:]]*([0-9]+\.[0-9]+\.[0-9]+) ]]; then
    target_version="${match[1]}"
  # 对于 ^X.Y.Z 格式
  elif [[ "$node_range" =~ ^\^([0-9]+\.[0-9]+\.[0-9]+) ]]; then
    target_version="${match[1]}"
  # 对于 ~X.Y.Z 格式
  elif [[ "$node_range" =~ ^\~([0-9]+\.[0-9]+\.[0-9]+) ]]; then
    target_version="${match[1]}"
  # 对于 X.Y.Z 格式
  elif [[ "$node_range" =~ ^([0-9]+\.[0-9]+\.[0-9]+)$ ]]; then
    target_version="${match[1]}"
  # 对于 X.Y.x 格式
  elif [[ "$node_range" =~ ^([0-9]+\.[0-9]+)\.x$ ]]; then
    target_version="${match[1]}.0"
  fi

  # 检查当前版本是否满足要求
  if [[ -n "$current_version" && -n "$target_version" ]]; then
    # 拆分当前版本号进行比较
    local current_major=$(echo "$current_version" | cut -d. -f1)
    local current_minor=$(echo "$current_version" | cut -d. -f2)
    local current_patch=$(echo "$current_version" | cut -d. -f3)
    local target_major=$(echo "$target_version" | cut -d. -f1)
    local target_minor=$(echo "$target_version" | cut -d. -f2)
    local target_patch=$(echo "$target_version" | cut -d. -f3)
    
    local current_satisfies=false
    
    # 对于 >=X.Y.Z 格式，检查当前版本是否满足要求
    if [[ "$node_range" =~ ^\>\= ]]; then
      if [[ "$current_major" -gt "$target_major" ]]; then
        current_satisfies=true
      elif [[ "$current_major" -eq "$target_major" ]]; then
        if [[ "$current_minor" -gt "$target_minor" ]]; then
          current_satisfies=true
        elif [[ "$current_minor" -eq "$target_minor" ]]; then
          if [[ "$current_patch" -ge "$target_patch" ]]; then
            current_satisfies=true
          fi
        fi
      fi
    else
      # 对于其他格式，检查当前版本是否匹配
      if [[ "$current_version" == "$target_version"* ]]; then
        current_satisfies=true
      fi
    fi
    
    # 如果当前版本满足要求，直接返回
    if [[ "$current_satisfies" == "true" ]]; then
      return 0
    fi
  fi

  # 获取已安装的 Node.js 版本列表
  local installed_versions=$(nvm ls --no-alias | grep -oE 'v[0-9]+\.[0-9]+\.[0-9]+' | sed 's/^v//' | sort -V)

  # 查找满足条件的版本
  local found_version=""
  if [[ -n "$target_version" ]]; then
    # 对于 >=X.Y.Z，查找大于等于该版本的最新版本
    if [[ "$node_range" =~ ^\>\= ]]; then
      while IFS= read -r version; do
        # 拆分版本号进行比较
        local current_major=$(echo "$version" | cut -d. -f1)
        local current_minor=$(echo "$version" | cut -d. -f2)
        local current_patch=$(echo "$version" | cut -d. -f3)
        local target_major=$(echo "$target_version" | cut -d. -f1)
        local target_minor=$(echo "$target_version" | cut -d. -f2)
        local target_patch=$(echo "$target_version" | cut -d. -f3)
        
        # 比较主版本号
        if [[ "$current_major" -gt "$target_major" ]]; then
          found_version="$version"
        elif [[ "$current_major" -eq "$target_major" ]]; then
          # 比较次版本号
          if [[ "$current_minor" -gt "$target_minor" ]]; then
            found_version="$version"
          elif [[ "$current_minor" -eq "$target_minor" ]]; then
            # 比较修订版本号
            if [[ "$current_patch" -ge "$target_patch" ]]; then
              found_version="$version"
            fi
          fi
        fi
      done <<< "$installed_versions"
    else
      # 对于其他格式，直接查找匹配的版本
      while IFS= read -r version; do
        if [[ "$version" == "$target_version"* ]]; then
          found_version="$version"
        fi
      done <<< "$installed_versions"
    fi
  fi

  # 如果找到满足条件的版本，切换到该版本
  if [[ -n "$found_version" ]]; then
    echo -e "\033[34m🔄 切换到 Node.js v$found_version (满足 $node_range)\033[0m"
    nvm use "$found_version" >/dev/null 2>&1
    return $?
  fi

  # 如果没有找到满足条件的版本，尝试自动安装
  echo -e "\033[33m⚠️  未找到满足 $node_range 的已安装 Node.js 版本\033[0m" >&2

  # 尝试安装目标版本
  if [[ -n "$target_version" ]]; then
    echo -e "\033[36m📦 正在安装 Node.js v$target_version...\033[0m"
    if nvm install "$target_version" >/dev/null 2>&1; then
      echo -e "\033[32m✅ Node.js v$target_version 安装成功\033[0m"
      nvm use "$target_version" >/dev/null 2>&1
      return $?
    else
      echo -e "\033[31m❌ Node.js v$target_version 安装失败\033[0m" >&2
    fi
  fi

  echo -e "\033[31m❌ 无法自动切换到满足 $node_range 的 Node.js 版本\033[0m" >&2
  echo -e "\033[33m💡 请手动安装: nvm install --lts 或 nvm install <version>\033[0m" >&2
  return 1
}

# 添加到 chpwd 钩子，在切换目录时自动执行
autoload -U add-zsh-hook
add-zsh-hook chpwd auto_switch_node

# 在终端启动时也执行一次
auto_switch_node
```

### 使用说明

1. **在项目** `package.json` 中定义版本：

```json
{
  "engines": {
    "node": ">=20.11.0"
  }
}
```

1. **触发时机**：

   - **每次打开终端时**：自动检查当前目录的 package.json
   - **每次切换目录时**：自动检查新目录的 package.json
   - **手动执行**：在任何目录运行 `auto_switch_node` 命令

2. **版本切换逻辑**：

   - 如果当前版本满足要求 → 不执行任何操作
   - 如果当前版本不满足要求 → 自动切换到满足条件的最新版本
   - 如果没有满足条件的版本 → 尝试自动安装

### 特性

- ✅ **智能版本解析**：支持 `>=20.11.0`、`^18.0.0`、`~16.14.0`、精确版本和 `X.Y.x` 格式
- ✅ **最佳版本选择**：从已安装版本中选择最新的满足条件的版本
- ✅ **自动安装功能**：如果没有满足条件的版本，会尝试自动安装
- ✅ **清晰反馈**：提供彩色输出，显示切换状态和相关信息
- ✅ **错误处理**：完善的错误处理和用户提示
- ✅ **性能优化**：只在必要时进行版本切换，避免不必要的操作

### 工作原理

1. **检测阶段**：检查当前目录是否有 `package.json` 文件
2. **解析阶段**：读取并解析 `engines.node` 字段中的版本范围
3. **验证阶段**：检查当前 Node.js 版本是否满足要求
4. **查找阶段**：如果当前版本不满足，获取已安装版本列表并筛选满足条件的版本
5. **选择阶段**：从满足条件的版本中选择最新的一个
6. **切换阶段**：自动切换到选中的版本
7. **安装阶段**：如果没有满足条件的版本，尝试自动安装目标版本

### 示例场景

```bash
# 项目 A 要求 >=20.11.0，当前版本 v18.20.1
🔄 切换到 Node.js v24.11.1 (满足 >=20.11.0)

# 项目 B 要求 >=18.0.0，当前版本 v18.20.1
# (无输出，因为当前版本已满足要求)

# 项目 C 要求 ^14.17.2，当前版本 v18.20.1
🔄 切换到 Node.js v14.17.2 (满足 ^14.17.2)
```

### 重要说明

1. **zsh 正则表达式语法**：在 zsh 中，正则表达式需要使用引号包围，如 `"^\^([0-9]+\.[0-9]+\.[0-9]+)"`
2. **子 shell 限制**：在测试脚本或子 shell 中，`nvm use` 不会影响父 shell 的环境，这是正常行为
3. **实际使用**：在交互式终端中，版本切换会正常生效
4. **版本匹配逻辑**：
   - `>=X.Y.Z`：大于等于指定版本
   - `^X.Y.Z`：兼容指定版本的主版本，主版本号相同，次版本号可升级
   - `~X.Y.Z`：兼容指定版本的主次版本，主次版本号相同，补丁版本可升级
   - `X.Y.Z`：精确匹配
   - `X.Y.x`：匹配任意补丁版本

### 🔹 macOS

#### 1\. Homebrew

安装 [Homebrew](https://brew.sh/)，便于软件包管理。

#### 2\. Zsh 美化

推荐使用 [**Starship**](https://starship.rs/)：

```bash
brew install starship
```

zsh 配置：

```bash
# ~/.zshrc
# Starship 初始化
if command -v starship >/dev/null 2>&1; then
  eval "$(starship init zsh)"
fi
```

执行 `source ~/.zshrc` 使配置生效

##### 配置 Starship

创建 `~/.config/starship.toml` 文件并添加以下内容：

```toml
# ~/.config/starship.toml
format = """
$username\
$git_branch\
$nodejs\
$character"""
# 在提示符之间插入空行
add_newline = false

# 使用内置的username模块而不是env_var
[username]
style_user = "#17A8CD"
format = "[$user](#17A8CD):"
show_always = true

[git_branch]
# 显示当前的 Git 分支
format = "[$branch](#D670D6) "

[nodejs]
# 显示 Node.js 版本
format = "[$version](#23D184) "

[character]
# 确定提示符字符的显示，根据上次命令的成功或失败显示不同的颜色
success_symbol = "[➜](bold green)"
error_symbol = "[➜](bold red)"
```

此配置会显示用户名、Git分支、Node.js版本，并根据命令执行结果显示不同颜色的提示符。
