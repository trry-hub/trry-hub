最优方案：智能版本切换（推荐） ​

将以下代码添加到 ~/.zshrc：

bash
# =============================================================================
# 自动切换 Node.js 版本（基于 package.json engines.node）
# 优化点：
# 1. 缓存 package.json 解析结果，只在文件变更时重新读取
# 2. 直接扫描 $NVM_DIR/versions/node，避免每次执行 nvm ls
# 3. 使用 zsh 内建能力比较版本，减少 grep / sed / cut / sort 开销
# =============================================================================

autoload -U add-zsh-hook
zmodload zsh/stat 2>/dev/null

typeset -g __auto_node_cached_pkg=""
typeset -g __auto_node_cached_mtime=""
typeset -g __auto_node_cached_range=""

_auto_node_find_package_json() {
  local dir="$PWD"

  while [[ "$dir" != "/" ]]; do
    if [[ -f "$dir/package.json" ]]; then
      print -r -- "$dir/package.json"
      return 0
    fi
    dir="${dir:h}"
  done

  return 1
}

_auto_node_file_mtime() {
  local file="$1"
  local -a stat_result

  zstat -A stat_result +mtime -- "$file" 2>/dev/null || return 1
  print -r -- "${stat_result[1]}"
}

_auto_node_read_range() {
  local pkg_file="$1"

  if command -v jq >/dev/null 2>&1; then
    jq -r '.engines.node // empty' "$pkg_file" 2>/dev/null
    return 0
  fi

  tr -d '\n' < "$pkg_file" \
    | sed -nE 's/.*"engines"[[:space:]]*:[[:space:]]*\{[^}]*"node"[[:space:]]*:[[:space:]]*"([^"]+)".*/\1/p'
}

_auto_node_parse_range() {
  local range="${1//[[:space:]]/}"

  if [[ "$range" =~ '^>=([0-9]+\.[0-9]+\.[0-9]+)$' ]]; then
    print -r -- "gte:${match[1]}"
  elif [[ "$range" =~ '^\^([0-9]+\.[0-9]+\.[0-9]+)$' ]]; then
    print -r -- "caret:${match[1]}"
  elif [[ "$range" =~ '^~([0-9]+\.[0-9]+\.[0-9]+)$' ]]; then
    print -r -- "tilde:${match[1]}"
  elif [[ "$range" =~ '^([0-9]+\.[0-9]+\.[0-9]+)$' ]]; then
    print -r -- "exact:${match[1]}"
  elif [[ "$range" =~ '^([0-9]+\.[0-9]+)\.x$' ]]; then
    print -r -- "minor:${match[1]}"
  elif [[ "$range" =~ '^([0-9]+)\.x$' ]]; then
    print -r -- "major:${match[1]}"
  elif [[ "$range" =~ '^([0-9]+\.[0-9]+)$' ]]; then
    print -r -- "minor:${match[1]}"
  elif [[ "$range" =~ '^([0-9]+)$' ]]; then
    print -r -- "major:${match[1]}"
  fi
}

_auto_node_version_ge() {
  local left="$1"
  local right="$2"
  local -a left_parts right_parts
  local i left_part right_part

  left_parts=(${(s:.:)left})
  right_parts=(${(s:.:)right})

  for i in 1 2 3; do
    left_part=${left_parts[i]:-0}
    right_part=${right_parts[i]:-0}

    (( left_part > right_part )) && return 0
    (( left_part < right_part )) && return 1
  done

  return 0
}

_auto_node_satisfies() {
  local version="$1"
  local range="$2"
  local parsed mode base
  local -a version_parts base_parts

  parsed="$(_auto_node_parse_range "$range")"
  [[ -n "$parsed" ]] || return 1

  mode="${parsed%%:*}"
  base="${parsed#*:}"

  version_parts=(${(s:.:)version})
  base_parts=(${(s:.:)base})

  case "$mode" in
    gte)
      _auto_node_version_ge "$version" "$base"
      ;;
    caret)
      [[ "${version_parts[1]:-0}" == "${base_parts[1]:-0}" ]] || return 1
      _auto_node_version_ge "$version" "$base"
      ;;
    tilde)
      [[ "${version_parts[1]:-0}" == "${base_parts[1]:-0}" ]] || return 1
      [[ "${version_parts[2]:-0}" == "${base_parts[2]:-0}" ]] || return 1
      _auto_node_version_ge "$version" "$base"
      ;;
    exact)
      [[ "$version" == "$base" ]]
      ;;
    minor)
      [[ "${version_parts[1]:-0}.${version_parts[2]:-0}" == "$base" ]]
      ;;
    major)
      [[ "${version_parts[1]:-0}" == "$base" ]]
      ;;
    *)
      return 1
      ;;
  esac
}

_auto_node_find_best_installed() {
  local range="$1"
  local versions_dir="${NVM_DIR:-$HOME/.nvm}/versions/node"
  local best=""
  local dir version

  [[ -d "$versions_dir" ]] || return 1

  for dir in "$versions_dir"/v*(N/); do
    version="${dir:t}"
    version="${version#v}"

    _auto_node_satisfies "$version" "$range" || continue

    if [[ -z "$best" ]] || _auto_node_version_ge "$version" "$best"; then
      best="$version"
    fi
  done

  [[ -n "$best" ]] || return 1
  print -r -- "$best"
}

auto_switch_node() {
  local pkg_file pkg_mtime node_range current_version found_version parsed target_version

  pkg_file="$(_auto_node_find_package_json)" || return 0

  if ! command -v nvm >/dev/null 2>&1; then
    echo -e "\033[33m⚠️  nvm 未安装，无法自动切换 Node.js 版本\033[0m" >&2
    return 1
  fi

  pkg_mtime="$(_auto_node_file_mtime "$pkg_file" 2>/dev/null || true)"

  if [[ "$pkg_file" == "$__auto_node_cached_pkg" && "$pkg_mtime" == "$__auto_node_cached_mtime" ]]; then
    node_range="$__auto_node_cached_range"
  else
    node_range="$(_auto_node_read_range "$pkg_file")"
    node_range="${node_range//[[:space:]]/}"
    __auto_node_cached_pkg="$pkg_file"
    __auto_node_cached_mtime="$pkg_mtime"
    __auto_node_cached_range="$node_range"
  fi

  [[ -n "$node_range" ]] || return 0

  current_version="$(node -v 2>/dev/null)"
  current_version="${current_version#v}"

  if [[ -n "$current_version" ]] && _auto_node_satisfies "$current_version" "$node_range"; then
    return 0
  fi

  found_version="$(_auto_node_find_best_installed "$node_range")"
  if [[ -n "$found_version" ]]; then
    echo -e "\033[34m🔄 切换到 Node.js v$found_version (满足 $node_range)\033[0m"
    nvm use "$found_version" >/dev/null 2>&1
    return $?
  fi

  parsed="$(_auto_node_parse_range "$node_range")"
  target_version="${parsed#*:}"

  if [[ -n "$target_version" ]]; then
    echo -e "\033[36m📦 正在安装 Node.js v$target_version...\033[0m"
    if nvm install "$target_version" >/dev/null 2>&1; then
      echo -e "\033[32m✅ Node.js v$target_version 安装成功\033[0m"
      nvm use "$target_version" >/dev/null 2>&1
      return $?
    fi
    echo -e "\033[31m❌ Node.js v$target_version 安装失败\033[0m" >&2
  fi

  echo -e "\033[31m❌ 无法自动切换到满足 $node_range 的 Node.js 版本\033[0m" >&2
  echo -e "\033[33m💡 建议补充 .nvmrc，或手动执行 nvm install <version>\033[0m" >&2
  return 1
}

add-zsh-hook chpwd auto_switch_node

使用说明 ​

  1. 在项目 package.json 中定义版本：
json
{
  "engines": {
    "node": ">=20.11.0"
  }
}
1
2
3
4
5

  1. 触发时机：

    • 每次打开终端时：自动检查当前目录及上层目录中的 package.json
    • 每次切换目录时：自动检查当前目录所属项目的 package.json
    • 手动执行：在任何目录运行 auto_switch_node 命令

  2. 版本切换逻辑：

    • 如果当前版本满足要求 → 不执行任何操作
    • 如果当前版本不满足要求 → 自动切换到满足条件的最新版本
    • 如果没有满足条件的版本 → 尝试自动安装

特性 ​

  • ✅ 智能版本解析：支持 >=20.11.0、^18.0.0、~16.14.0、精确版本、20.x 和 20.11.x
  • ✅ 最佳版本选择：从已安装版本中选择最新的满足条件的版本
  • ✅ 自动安装功能：如果没有满足条件的版本，会尝试自动安装
  • ✅ 清晰反馈：提供彩色输出，显示切换状态和相关信息
  • ✅ 错误处理：完善的错误处理和用户提示
  • ✅ 性能优化：缓存 package.json 结果，并直接扫描 NVM 安装目录，避免高频调用 nvm ls

工作原理 ​

  1. 检测阶段：向上查找最近的 package.json，兼容在项目子目录中切换
  2. 解析阶段：读取并缓存 engines.node 字段，只有文件变更时才重新解析
  3. 验证阶段：检查当前 Node.js 版本是否满足要求
  4. 查找阶段：如果当前版本不满足，直接扫描 $NVM_DIR/versions/node 并筛选满足条件的版本
  5. 选择阶段：从满足条件的版本中选择最新的一个
  6. 切换阶段：自动切换到选中的版本
  7. 安装阶段：如果没有满足条件的版本，尝试自动安装目标版本

重要说明 ​

  1. zsh 正则表达式语法：在 zsh 中，正则表达式建议使用引号包围，如 "^\^([0-9]+\.[0-9]+\.[0-9]+)$"
  2. 子 shell 限制：在测试脚本或子 shell 中，nvm use 不会影响父 shell 的环境，这是正常行为
  3. 实际使用：在交互式终端中，版本切换会正常生效
  4. 性能建议：如果团队可以统一维护 .nvmrc，优先使用 .nvmrc + nvm use，它会更轻量
  5. 版本匹配逻辑：
    • >=X.Y.Z：大于等于指定版本
    • ^X.Y.Z：兼容指定版本的主版本，主版本号相同，次版本号可升级
    • ~X.Y.Z：兼容指定版本的主次版本，主次版本号相同，补丁版本可升级
    • X.Y.Z：精确匹配
    • X.Y.x / X.Y：匹配任意补丁版本
    • X.x / X：匹配任意次版本和补丁版本
